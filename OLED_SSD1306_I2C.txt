/*=========================================================
 * PIC18F4550 - Display OLED SSD1306 via I2C
 * Proyecto:    Interfaz visual con pantalla 128x64
 * Protocolo:   I2C (SSP module) - 100kHz
 * Direccion:   0x78 (write)
 *
 * Conexiones:
 *   SDA -> RB0
 *   SCL -> RB1
 *   Pull-ups 4.7k a VCC requeridos
 *
 * Entorno: MPLAB X + XC8 | Oscilador interno 8MHz
 *=========================================================*/

#pragma config FOSC   = INTOSC_EC
#pragma config WDT    = OFF
#pragma config LVP    = OFF
#pragma config PBADEN = OFF
#pragma config MCLRE  = OFF
#pragma config XINST  = OFF
#pragma config PWRT   = ON
#pragma config DEBUG  = OFF

#include <xc.h>
#define _XTAL_FREQ 8000000

/*--- Configuracion del display ---*/
#define DISPLAY_ADDR    0x78    /* Direccion I2C (escritura) */
#define CTRL_CMD        0x00    /* Byte control: comando     */
#define CTRL_DATA       0x40    /* Byte control: dato/pixel  */

/*--- Velocidad I2C ---
 * BRG = (Fosc / (4 * fscl)) - 1
 * BRG = (8MHz  / (4 * 100kHz)) - 1 = 19 */
#define BRG_VAL         19

/*=========================================================
 * CAPA I2C
 *=========================================================*/

static void bus_esperar(void){
    while(SSPSTATbits.BF || (SSPCON2 & 0x1F));
    __delay_us(5);
}

static void bus_inicio(unsigned char direccion){
    bus_esperar();
    SSPCON2bits.SEN = 1;
    while(SSPCON2bits.SEN);
    SSPBUF = direccion;
    while(!PIR1bits.SSPIF);
    PIR1bits.SSPIF = 0;
}

static void bus_enviar(unsigned char byte){
    bus_esperar();
    SSPBUF = byte;
    while(!PIR1bits.SSPIF);
    PIR1bits.SSPIF = 0;
}

static void bus_fin(void){
    bus_esperar();
    SSPCON2bits.PEN = 1;
    while(SSPCON2bits.PEN);
    PIR1bits.SSPIF = 0;
}

void bus_init(void){
    TRISBbits.TRISB0 = 1;   /* SDA: entrada */
    TRISBbits.TRISB1 = 1;   /* SCL: entrada */

    SSPSTAT  = 0x80;        /* Slew rate OFF              */
    SSPCON1  = 0x28;        /* Master I2C, SSPEN activo   */
    SSPCON2  = 0x00;
    SSPADD   = BRG_VAL;
    PIR1bits.SSPIF = 0;
}

/*=========================================================
 * CAPA SSD1306
 *=========================================================*/

static void display_cmd(unsigned char c){
    bus_inicio(DISPLAY_ADDR);
    bus_enviar(CTRL_CMD);
    bus_enviar(c);
    bus_fin();
}

static void display_pixel(unsigned char d){
    bus_inicio(DISPLAY_ADDR);
    bus_enviar(CTRL_DATA);
    bus_enviar(d);
    bus_fin();
}

void display_borrar(void){
    unsigned char pg, col;
    for(pg = 0; pg < 8; pg++){
        display_cmd(0xB0 | pg);
        display_cmd(0x00);
        display_cmd(0x10);
        for(col = 0; col < 128; col++)
            display_pixel(0x00);
    }
}

void display_cursor(unsigned char pg, unsigned char col){
    display_cmd(0xB0 | pg);
    display_cmd(col & 0x0F);
    display_cmd(0x10 | ((col >> 4) & 0x0F));
}

void display_init(void){
    __delay_ms(100);

    display_cmd(0xAE);  /* Apagar display            */
    display_cmd(0xD5);  /* Clock / divisor           */
    display_cmd(0x80);
    display_cmd(0xA8);  /* Multiplex                 */
    display_cmd(0x3F);  /* 64 lineas                 */
    display_cmd(0xD3);  /* Offset vertical           */
    display_cmd(0x00);
    display_cmd(0x40);  /* Linea de inicio = 0       */
    display_cmd(0x8D);  /* Charge pump               */
    display_cmd(0x14);  /* Habilitar                 */
    display_cmd(0x20);  /* Modo de direccionamiento  */
    display_cmd(0x00);  /* Horizontal                */
    display_cmd(0xA1);  /* SEG remap normal          */
    display_cmd(0xC8);  /* COM scan direction        */
    display_cmd(0xDA);  /* COM pins config           */
    display_cmd(0x12);
    display_cmd(0x81);  /* Contraste                 */
    display_cmd(0xCF);
    display_cmd(0xD9);  /* Pre-charge period         */
    display_cmd(0xF1);
    display_cmd(0xDB);  /* VCOMH                     */
    display_cmd(0x40);
    display_cmd(0xA4);  /* Contenido desde RAM       */
    display_cmd(0xA6);  /* Modo normal (no invertido)*/
    display_cmd(0xAF);  /* Encender display          */

    __delay_ms(10);
    display_borrar();
}

static void escribir_char(unsigned char c){
    unsigned char k;
    if(c < 32 || c > 90) c = 32;
    for(k = 0; k < 5; k++)
        display_pixel(FUENTE[c - 32][k]);
    display_pixel(0x00);        /* Separador entre caracteres */
}

void escribir_texto(unsigned char pg, unsigned char col,
                    const char *msg){
    display_cursor(pg, col);
    while(*msg)
        escribir_char(*msg++);
}

/*=========================================================
 * PROGRAMA PRINCIPAL
 *=========================================================*/
void main(void){

    OSCCON = 0x72;      /* Oscilador interno a 8MHz   */
    ADCON1 = 0x0F;      /* Puerto B completamente digital */

    bus_init();         /* Iniciar modulo I2C         */
    display_init();     /* Iniciar pantalla OLED      */

    /* Contenido en pantalla
     * Rango de paginas: 0-7  |  Columnas: 0-127       */
    escribir_texto(0, 0, "Display OLED SSD1306 via I2C ");
    escribir_texto(2, 0, "...");
    escribir_texto(4, 0, "PIC18F4550");
    escribir_texto(6, 0, "...");

    for(;;);            /* Loop infinito              */
}
