/*=========================================================
 * PIC18F4550 - Sensor LDR + Display OLED SSD1306
 * Proyecto:    Lectura de luminosidad con LDR
 * Sensor:      LDR en divisor de tension (entrada ADC)
 * Display:     OLED 128x64 via I2C (0x78)
 *
 * Funcionamiento del LDR:
 *   El LDR varia su resistencia segun la luz:
 *   - Mucha luz  -> resistencia baja  -> voltaje alto en AN0
 *   - Poca luz   -> resistencia alta  -> voltaje bajo en AN0
 *   Se lee con el ADC de 10 bits (0-1023) y se convierte
 *   a porcentaje de luminosidad (0-100%)
 *
 * Circuito divisor de tension:
 *   VCC (5V)
 *     |
 *    LDR
 *     |
 *     +-----> AN0 (RA0) - entrada ADC
 *     |
 *   10k?
 *     |
 *    GND
 *
 * Conexiones:
 *   RA0      -> Divisor LDR/10k?
 *   RB0 SDA  -> SDA OLED
 *   RB1 SCL  -> SCL OLED
 *   Pull-ups 4.7k en SDA y SCL
 *
 * Entorno: MPLAB X + XC8 | Oscilador interno 8MHz
 *=========================================================*/

#pragma config FOSC   = INTOSC_EC
#pragma config WDT    = OFF
#pragma config LVP    = OFF
#pragma config PBADEN = OFF
#pragma config MCLRE  = OFF
#pragma config XINST  = OFF
#pragma config PWRT   = ON
#pragma config DEBUG  = OFF

#include <xc.h>
#define _XTAL_FREQ 8000000

/*--- Display OLED SSD1306 ---*/
#define DISPLAY_ADDR    0x78
#define CTRL_CMD        0x00
#define CTRL_DATA       0x40

/*--- ADC ---*/
#define ADC_MAX         1023    /* 10 bits: 0-1023              */

/*--- Velocidad I2C ---*/
#define BRG_VAL         19      /* 100kHz a 8MHz                */

/*=========================================================
 * CAPA I2C
 *=========================================================*/

static void bus_esperar(void){
    while(SSPSTATbits.BF || (SSPCON2 & 0x1F));
    __delay_us(5);
}

static void bus_inicio(unsigned char direccion){
    bus_esperar();
    SSPCON2bits.SEN = 1;
    while(SSPCON2bits.SEN);
    SSPBUF = direccion;
    while(!PIR1bits.SSPIF);
    PIR1bits.SSPIF = 0;
}

static void bus_enviar(unsigned char byte){
    bus_esperar();
    SSPBUF = byte;
    while(!PIR1bits.SSPIF);
    PIR1bits.SSPIF = 0;
}

static void bus_fin(void){
    bus_esperar();
    SSPCON2bits.PEN = 1;
    while(SSPCON2bits.PEN);
    PIR1bits.SSPIF = 0;
}

void bus_init(void){
    TRISBbits.TRISB0 = 1;
    TRISBbits.TRISB1 = 1;
    SSPSTAT  = 0x80;
    SSPCON1  = 0x28;
    SSPCON2  = 0x00;
    SSPADD   = BRG_VAL;
    PIR1bits.SSPIF = 0;
}

/*=========================================================
 * CAPA ADC
 * El LDR se conecta a RA0 = canal AN0
 * El ADC del PIC18F4550 es de 10 bits -> 0 a 1023
 *=========================================================*/

void adc_init(void){
    /* RA0 como entrada analogica, resto digital */
    TRISA  = 0xFF;          /* Puerto A como entrada            */
    ADCON1 = 0x0E;          /* Solo AN0 analogico, resto digital*/

    /* Configuracion ADC:
     * ADCON0: canal AN0, ADC ON
     * ADCON2: Fosc/32, 12TAD, justificado a la derecha        */
    ADCON0 = 0x01;          /* CHS=0000 (AN0), ADON=1           */
    ADCON2 = 0xAA;          /* ADCS=010(Fosc/32), ACQT=101(12TAD)
                             * ADFM=1 (derecha)                 */
    __delay_ms(1);          /* Estabilizacion del ADC           */
}

/* Realiza una conversion ADC y retorna el resultado (0-1023) */
unsigned int adc_leer(void){
    ADCON0bits.GO = 1;          /* Inicia conversion            */
    while(ADCON0bits.GO);       /* Espera que termine           */
    return ((unsigned int)(ADRESH << 8) | ADRESL);
}

/* Convierte lectura ADC a porcentaje de luminosidad (0-100)
 * Usa regla de tres simple: pct = (adc * 100) / 1023
 * Se usa unsigned long para evitar desbordamiento            */
unsigned char adc_a_porcentaje(unsigned int adc){
    return (unsigned char)(((unsigned long)adc * 100UL) / ADC_MAX);
}

/*=========================================================
 * CAPA OLED - Identica a proyectos anteriores
 *=========================================================*/

static void display_cmd(unsigned char c){
    bus_inicio(DISPLAY_ADDR);
    bus_enviar(CTRL_CMD);
    bus_enviar(c);
    bus_fin();
}

static void display_pixel(unsigned char d){
    bus_inicio(DISPLAY_ADDR);
    bus_enviar(CTRL_DATA);
    bus_enviar(d);
    bus_fin();
}

void display_borrar(void){
    unsigned char pg, col;
    for(pg = 0; pg < 8; pg++){
        display_cmd(0xB0 | pg);
        display_cmd(0x00);
        display_cmd(0x10);
        for(col = 0; col < 128; col++)
            display_pixel(0x00);
    }
}

void display_cursor(unsigned char pg, unsigned char col){
    display_cmd(0xB0 | pg);
    display_cmd(col & 0x0F);
    display_cmd(0x10 | ((col >> 4) & 0x0F));
}

void display_init(void){
    __delay_ms(100);
    display_cmd(0xAE);
    display_cmd(0xD5); display_cmd(0x80);
    display_cmd(0xA8); display_cmd(0x3F);
    display_cmd(0xD3); display_cmd(0x00);
    display_cmd(0x40);
    display_cmd(0x8D); display_cmd(0x14);
    display_cmd(0x20); display_cmd(0x00);
    display_cmd(0xA1);
    display_cmd(0xC8);
    display_cmd(0xDA); display_cmd(0x12);
    display_cmd(0x81); display_cmd(0xCF);
    display_cmd(0xD9); display_cmd(0xF1);
    display_cmd(0xDB); display_cmd(0x40);
    display_cmd(0xA4);
    display_cmd(0xA6);
    display_cmd(0xAF);
    __delay_ms(10);
    display_borrar();
}

static void escribir_char(unsigned char c){
    unsigned char k;
    if(c < 32 || c > 90) c = 32;
    for(k = 0; k < 5; k++)
        display_pixel(FUENTE[c - 32][k]);
    display_pixel(0x00);
}

void escribir_texto(unsigned char pg, unsigned char col,
                    const char *msg){
    display_cursor(pg, col);
    while(*msg)
        escribir_char(*msg++);
}

static void escribir_numero(unsigned char pg, unsigned char col,
                             unsigned int num, unsigned char digitos){
    char buf[6];
    signed char i;
    for(i = digitos - 1; i >= 0; i--){
        buf[i] = '0' + (num % 10);
        num   /= 10;
    }
    buf[digitos] = '\0';
    escribir_texto(pg, col, buf);
}

/*=========================================================
 * BARRA DE LUMINOSIDAD
 * Dibuja una barra grafica proporcional al porcentaje
 * en la pagina indicada. Ancho maximo = 120 pixeles.
 * Cada pixel de ancho = 1.2% de luminosidad
 *=========================================================*/
void dibujar_barra(unsigned char pg, unsigned char pct){
    unsigned char ancho, col;

    /* Calculo ancho proporcional: max 120px = 100% */
    ancho = (unsigned char)(((unsigned int)pct * 120U) / 100U);

    display_cursor(pg, 4);
    for(col = 0; col < 120; col++){
        if(col < ancho)
            display_pixel(0x3C); /* Barra llena: 0011 1100      */
        else
            display_pixel(0x00); /* Barra vacia                 */
    }
}

/*=========================================================
 * PROGRAMA PRINCIPAL
 *=========================================================*/
void main(void){
    unsigned int  lectura_adc;      /* Valor crudo 0-1023        */
    unsigned char luminosidad;      /* Porcentaje 0-100          */
    unsigned char nivel[5];         /* Texto del nivel de luz    */

    OSCCON = 0x72;          /* Oscilador interno 8MHz            */

    bus_init();             /* Iniciar modulo I2C                */
    adc_init();             /* Iniciar ADC en AN0                */
    display_init();         /* Iniciar OLED                      */

    /* --- Encabezado fijo --- */
    escribir_texto(0, 0, "LDR - LUZ");
    escribir_texto(1, 0, "UNICAUCA");
    escribir_texto(3, 0, "LUZ");
    escribir_texto(3, 60, "PCT");
    escribir_texto(5, 0, "NIVEL");
    escribir_texto(7, 0, "ADC");

    for(;;){
        lectura_adc  = adc_leer();
        luminosidad  = adc_a_porcentaje(lectura_adc);

        /* Porcentaje numerico */
        escribir_numero(4, 0,  luminosidad, 3);

        /* Barra grafica proporcional */
        dibujar_barra(6, luminosidad);

        /* Nivel cualitativo segun rango */
        if(luminosidad >= 75){
            escribir_texto(5, 36, "ALTO  ");
        } else if(luminosidad >= 40){
            escribir_texto(5, 36, "MEDIO ");
        } else {
            escribir_texto(5, 36, "BAJO  ");
        }

        /* Valor ADC crudo (util para calibrar) */
        escribir_numero(7, 30, lectura_adc, 4);

        __delay_ms(300);    /* Actualizo cada 300ms              */
    }
}
